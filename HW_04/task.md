typedef int semaphore;
semaphore mutex = 1;
semaphore db = 1;
int rc = 1;

void reader(void) {
    while(TRUE) {
        down(&mutex);
        rc++;
        up(&mutex);
        read_data_base();
        down(&mutex);
        rc--;
        if (rc == 0) up(&db);
        use_data_read();
    }
}
void writer(void) {
    while(TRUE) {
        think_up_data();
        down(&db);
        write_data_base();
        up(&db);
    }
}
